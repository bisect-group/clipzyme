# ESM-2 weights: copy the read-only torch_hub folder from emulator_old_data (Clipzyme/emulator_bench/cache/torch_hub)
# to outputs/baselines/Clipzyme/weights/torch_hub.

# 0. Check which runs exist on disk (no work is done)
$PY -m emulator_bench.run_all --gpus 0 --dry_run

# 1. All splits found on disk x seeds 666 777 888, on GPUs 0 1 2 (one run per GPU)
#    Prepare runs first on the CPU. Train and evaluate follow. The summary CSVs are rebuilt at the end.
$PY -m emulator_bench.run_all --gpus 0 1 2

# A 23 GB GPU needs --checkpoint_esm for batch 8
$PY -m emulator_bench.run_all --gpus 0 1 2 --checkpoint_esm

# Exact speed-ups (SDPA, length-sorted ESM chunks, adaptive recomputation, vectorised reverse edges); also applies to evaluate
$PY -m emulator_bench.run_all --gpus 0 1 2 --fast

# Same, with 2 runs per GPU and --amp
$PY -m emulator_bench.run_all --gpus 0 1 2 --runs_per_gpu 2 --amp

# Frozen ESM-2 (configs/train/clip_esm_frozen.json): build the ESM cache first (one process per GPU, resumable), then run
CUDA_VISIBLE_DEVICES=0 $PY -m emulator_bench.esm_cache --shard 0 --shards 3
CUDA_VISIBLE_DEVICES=1 $PY -m emulator_bench.esm_cache --shard 1 --shards 3
CUDA_VISIBLE_DEVICES=2 $PY -m emulator_bench.esm_cache --shard 2 --shards 3
$PY -m emulator_bench.run_all --gpus 0 1 2 --runs_per_gpu 2 --frozen_esm --fast
# --epochs N replaces the epoch count (10 with --frozen_esm, 20 otherwise)

# Equivalence check: cached path against the original frozen path on a small subset (needs one GPU)
CUDA_VISIBLE_DEVICES=1 $PY -m emulator_bench.check_equivalence --work_dir /path/to/scratch

# One split group (all its thresholds), one seed
$PY -m emulator_bench.run_all --gpus 0 --split_groups enzyme_sequence_splits --seeds 666

# One split by hand: prepare, train, evaluate
$PY -m emulator_bench.dataset_adapter --split_groups random_splits_grouped_rxn_key/all
CUDA_VISIBLE_DEVICES=0 $PY -m emulator_bench.train --split_groups random_splits_grouped_rxn_key/all --seeds 666
CUDA_VISIBLE_DEVICES=0 $PY -m emulator_bench.evaluate --split_groups random_splits_grouped_rxn_key/all --seeds 666

# Rebuild the summary CSVs from the saved predictions
$PY -m emulator_bench.results
